terminate_shell_shard() {
trap '' TERM
diagnostic="${RUNNER_TEMP:-/tmp}/shell-hang-${ONTOS_CI_WORKER_SLOT}-$$.diag.log"

if [[ -n "$task_pid" ]]; then
kill -TERM -- "-$task_pid" 2>/dev/null || true
timeout --kill-after=1s 3s tail --pid="$task_pid" -f /dev/null || true
kill -KILL -- "-$task_pid" 2>/dev/null || true
fi
exit 124
}

parallel_tasks=()
for task in "${tasks[@]}"; do
if [[ " ${exclusive_tasks[*]} " == *" $task "* ]]; then
ONTOS_CI_WORKER_SLOT=0 bash scripts/ci/run-integration-suites.sh --task "$task" || status=$?
else
parallel_tasks+=("$task")
fi
done
if [[ ${#parallel_tasks[@]} -gt 0 ]]; then
printf '%s\0' "${parallel_tasks[@]}" |
xargs -0 --max-procs="$lanes" --max-args=1 --process-slot-var=ONTOS_CI_WORKER_SLOT        bash scripts/ci/run-integration-suites.sh --task || status=$?
